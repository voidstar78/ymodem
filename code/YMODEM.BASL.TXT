REM 5678901234567890123456789012 XIPHOD - 2026 89012345678901234567890
REM      1         2         3         4         5         6         7
REM YMODEM BATCH
REM X16 SERIAL (CTS/RTS ENABLED VERSION) - XIPHOD JANUARY 2025

SCREEN 0
PRINT "{WHITE}";

SERIAL.IN% = 0 : REM DEFINE THIS FIRST FOR PERFORMANCE

MAGIC.VALUE% = $1021 : REM FOR 16-BIT CRC

DIM BAUD.DIV.HI(22)
DIM BAUD.DIV.LO(22)
DIM BAUD.RATE$(22)
RESTORE BAUD.DIVS
FOR I = 1 TO 22
  READ HI:READ LO:READ RATE$
  BAUD.DIV.HI(I) = HI
  BAUD.DIV.LO(I) = LO
  BAUD.RATE$(I) = RATE$
NEXT I
BAUD.CURR = 7 : REM DEFAULT TO 1200 BAUD

REM FOR PURPOSES OF TEST, THESE ARE FIXED.  "REAL" PROGRAM
REM CAN AUTO-DETECT THESE OR LET USER SPECIFY.
REM ----------------
DEVICE.MODE=1 : REM 1 == SERIAL, 0 == "NETWORK" WIFI
PORT.IDX=4 : REM PORT 7
PORT.STATE=0 : REM 0 = LOW ADDRESSES, 1 = HIGH ADDRESSES

DIM SERIAL.ADDR(14)
GOSUB INIT.SERIAL.ADDR

START.TERM:
SERIAL.REG.BASE = SERIAL.ADDR(7*PORT.STATE+PORT.IDX)
PRINT "PORT ADDR ";HEX$(SERIAL.REG.BASE)

REM **************************************************

GOSUB INIT.SERIAL.REGS
GOSUB SERIAL.INIT

REM 4K INPUT BUFFER
IN.BUFFER.START.ADDRESS = $A000
IN.BUFFER.FINISH.ADDRESS = $B000

IN.BUFFER.POS = IN.BUFFER.START.ADDRESS

BANK 2 : REM USE BANK 1 FOR THE RECEIVE BUFFER

POKE REG.WRITE.MCR, %00100011 : REM SET AUTO-CTS AND AUTO-RTS AND DTR

CURR.EXPECTED.MARKER1% = $00 : REM BLOCK COUNT
CURR.EXPECTED.MARKER2% = $FF : REM INVERSE COUNT  (MARKER1+MARKR2=255)
CURR.BLOCK = 0

BLOCK.HEADER.FN$ = ""  
BLOCK.HEADER.LEN$ = ""
BLOCK.HEADER.LEN = 0 : REM EXPECTED TOTAL NUMBER OF BYTES FOR CURRENT FILE
RECEIVED.BYTES = 0 : REM BYTES THAT APPLY TO THE FILE BEING SENT (NOT TOTAL SERIAL BYTES)

GOSUB WAIT.LINE.QUIET

REM SEND "C" TO GET ATTENTION OF SENDER
A$ = "C" : REM ASCII HEX $43
GOSUB SEND.ONE

REM TEST.QUERY:
REM   LSR% = PEEK(REG.READ.LSR) : IF (LSR% AND 1) = 0 THEN GOTO TEST.QUERY
REM   SERIAL.IN% = PEEK(REG.READ.RBR)  
REM   PRINT " ";HEX$(SERIAL.IN%)
REM   GOTO TEST.QUERY
REM SKIP.1:

CRC.RESULT = 0 : REM INITIAL VALUE (XMODEM INIT 0, FOR IBM 3740 - START WITH $FFFF)
REM WARNING: DON'T USE % TYPE IN CRC.RESULT

REM THESE ONLY WORK WHEN X <= 32767 AND X >= -32768 (SPECIFICALLY THE "OR" AND "AND" KEYWORDS)
REM X XOR Y = (X OR Y) AND NOT (X AND Y)
REM ALTERNATIVE: (X+Y) - 2*(X AND Y)
DEF FN XM(X) = (X + XV) - 2*(X AND XV)

REM STAGE1 - WAIT FOR #01(128 BYTE BLOCK) OR #02(1024 BYTE BLOCK)
REM BETWEEN BLOCKS, A #04 COULD BE RECEIVED INDICATING END-OF-TRANSFER
MAIN.LOOP.STAGE1:
  LSR% = PEEK(REG.READ.LSR) : IF (LSR% AND 1)=0 THEN MAIN.LOOP.STAGE1  
  SERIAL.IN% = PEEK(REG.READ.RBR)  
  IF (SERIAL.IN% = 0) THEN GOTO MAIN.LOOP.STAGE1  
  PRINT "  1:";HEX$(SERIAL.IN%);" ";
  IF (SERIAL.IN% = 1) THEN BLOCK.SIZE% = 128: GOTO MAIN.LOOP.STAGE2
  IF (SERIAL.IN% = 2) THEN BLOCK.SIZE% = 1024: GOTO MAIN.LOOP.STAGE2
  IF (SERIAL.IN% = 4) THEN GOSUB HANDLE.EOT
  GOTO MAIN.LOOP.STAGE1

REM STAGE2 - WAIT FOR "EXPECTED BLOCK MARKER #1"
MAIN.LOOP.STAGE2:
  LSR% = PEEK(REG.READ.LSR) : IF (LSR% AND 1)=0 THEN MAIN.LOOP.STAGE2
  SERIAL.IN% = PEEK(REG.READ.RBR) 
  IF (SERIAL.IN% = CURR.EXPECTED.MARKER1%) THEN PRINT "  2:";HEX$(SERIAL.IN%);" ";:GOTO MAIN.LOOP.STAGE3
  PRINT " BAD MARKER 1: ";HEX$(SERIAL.IN%)
  GOTO MAIN.LOOP.STAGE1 : REM START OVER...

REM STAGE3 - WAIT FOR "EXPECTED BLOCK MARKER #2"
MAIN.LOOP.STAGE3:
  LSR% = PEEK(REG.READ.LSR) : IF (LSR% AND 1)=0 THEN MAIN.LOOP.STAGE3
  SERIAL.IN% = PEEK(REG.READ.RBR) 
  PRINT "  3:";HEX$(SERIAL.IN%):GOTO MAIN.LOOP.STAGE4.PREP
  IF (SERIAL.IN% = CURR.EXPECTED.MARKER2%) THEN PRINT "  3.2:";HEX$(SERIAL.IN%);" ":GOTO MAIN.LOOP.STAGE4.PREP
  
  REM OR CAN REPEAT MARKER1 AGAIN...
  REM IF (SERIAL.IN% = CURR.EXPECTED.MARKER1%) THEN PRINT "  3.1:";HEX$(SERIAL.IN%);" ":GOTO MAIN.LOOP.STAGE4.PREP  
  
  REM PRINT " BAD MARKER 2: ";HEX$(SERIAL.IN%)
  GOTO MAIN.LOOP.STAGE1 : REM START OVER...

REM STAGE4 - BUFFER DOWN THE DATA PAYLOAD
REM ALSO ACCUMULATE/CALCULATE CRC AS WE GO (NECESSARY TO AVOID A "STALL"
REM LATER IN TRYING TO POST-CALCULATE THE CRC ON THE BLOCK)
MAIN.LOOP.STAGE4.PREP:
  PRINT "4:";
  IF CURR.BLOCK = 0 THEN PRINT "{YELLOW}HEADER{WHITE}";
  IF CURR.BLOCK <> 0 THEN PRINT "{GREEN}DATA{WHITE}";
  PRINT " BLOCK ";HEX$(CURR.EXPECTED.MARKER1%);"/";HEX$(CURR.EXPECTED.MARKER2%);" SIZE ";BLOCK.SIZE%;"B ";
MAIN.LOOP.STAGE4:  
  REM THIS COULD BE A HEADER BLOCK OR A DATA BLOCK, EITHER WAY SAME ACTION...
  IF (IN.BUFFER.POS-IN.BUFFER.START.ADDRESS) = BLOCK.SIZE% THEN PRINT "[ BLOCK {BLACK}"; : GOTO MAIN.LOOP.STAGE5
  LSR% = PEEK(REG.READ.LSR) : IF (LSR% AND 1)=0 THEN MAIN.LOOP.STAGE4
  SERIAL.IN% = PEEK(REG.READ.RBR)
  PRINT HEX$(SERIAL.IN%);
  POKE IN.BUFFER.POS, SERIAL.IN% : IN.BUFFER.POS = IN.BUFFER.POS+1  
  REM ACCUMULATE BYTE INTO 16-BIT CRC RESULT...
  XV = SERIAL.IN% * 256 : REM (CRC.DATA AT CURRENT INDEX) << 8
  REM THE FOLLOWING IS NECESSARY SO WE CAN USE "XV" LATER IN THE DO.XOR SUBROUTINE
  IF XV > 32767 THEN XV = XV - 65536
  IF XV < -32768 THEN XV = XV + 65536
  CRC.TEMP = CRC.RESULT : CRC.TEMP = FN XM(CRC.TEMP) : CRC.RESULT = CRC.TEMP : XV = MAGIC.VALUE% : REM PREPARE THAT WE WILL XOR BY THE XMODEM POLYNOMIAL (IN HIGH.BIT CASE)
  FOR I = 0 TO 7
    REM CAN'T USE "AND" RELIABLY ON SIGNED 16-BIT, BUT WE JUST NEED
    REM TO CHECK IF THE HIGH BIT (MSB) IS SET. SO JUST LOOK AT THE HIGH-BYTE
    REM BY DIVIDING THE VALUE BY 256 (SAME AS SHR >> 8).
    CRC.LOW% = CRC.RESULT / 256 : IF CRC.LOW% AND $80 THEN GOTO HANDLE.HIGH.BIT.CASE
    REM ELSE...
    CRC.RESULT = CRC.RESULT * 2 : REM (CRC << 1)
HIGH.BIT.WAS.SET:
    REM WRAP CRC.RESULT TO REMAIN COMPATIBLE WITH SIGNED LOGIC OPERATIONS
    IF CRC.RESULT > 32767 THEN CRC.RESULT = CRC.RESULT - 65536
    IF CRC.RESULT < -32768 THEN CRC.RESULT = CRC.RESULT + 65536    
  NEXT
  GOTO MAIN.LOOP.STAGE4 : REM REPEAT FOR THE NEXT DATA BYTE BEING SENT

HANDLE.HIGH.BIT.CASE:
  CRC.TEMP = CRC.RESULT * 2 : REM (CRC << 1)
  IF CRC.TEMP > 32767 THEN CRC.TEMP = CRC.TEMP - 65536
  IF CRC.TEMP < -32768 THEN CRC.TEMP = CRC.TEMP + 65536
  CRC.TEMP = FN XM(CRC.TEMP) : REM XOR'D WITH XV
  CRC.RESULT = CRC.TEMP : GOTO HIGH.BIT.WAS.SET : REM RESUME PAST THE IF-ELSE THAT WE CAME FROM    

REM STAGE5 - OBTAIN 16-BIT (2 BYTE) CRC FROM SENDER, SEE IF MATCHES LOCAL CRC.RESULT
MAIN.LOOP.STAGE5:
  PRINT "{WHITE} ]"
  REM PRINT HEX$(PEEK(IN.BUFFER.START.ADDRESS));" ";HEX$(PEEK(IN.BUFFER.START.ADDRESS+1));" ";HEX$(PEEK(IN.BUFFER.START.ADDRESS+2));" ";HEX$(PEEK(IN.BUFFER.START.ADDRESS+3));"...";
  REM I = IN.BUFFER.START.ADDRESS + BLOCK.SIZE%
  REM PRINT HEX$(PEEK(I-4));" ";HEX$(PEEK(I-3));" ";HEX$(PEEK(I-2));" ";HEX$(PEEK(I-1));"{WHITE}]"
  
  IN.BUFFER.POS = IN.BUFFER.START.ADDRESS : REM RESET WORKING BUFFER POSITION INDEX BACK TO BEGINNING
  
  CRC.TEMP = CRC.RESULT : IF CRC.TEMP < 0 THEN CRC.TEMP = CRC.TEMP + 65536
  PRINT "5:CRC LOCAL [{CYAN}";CRC.TEMP;"{WHITE}] / REMOTE [{CYAN}";

MAIN.LOOP.STAGE5.A:
  LSR% = PEEK(REG.READ.LSR) : IF (LSR% AND 1)=0 THEN MAIN.LOOP.STAGE5.A
  CRC.RECV = PEEK(REG.READ.RBR) * 256 : REM HIGH BYTE

MAIN.LOOP.STAGE5.B:
  LSR% = PEEK(REG.READ.LSR) : IF (LSR% AND 1)=0 THEN MAIN.LOOP.STAGE5.B
  CRC.RECV = CRC.RECV + PEEK(REG.READ.RBR) : REM ADD IN LOW BYTE

  REM SCALE CRC.RECV TO MATCH RANGE WE ARE USING FOR INTERNAL CRC.RESULT
  IF CRC.RECV < 0 THEN CRC.RECV = CRC.RECV + 65536

  PRINT CRC.RECV;"{WHITE}]"

  IF (CRC.TEMP = CRC.RECV) THEN GOTO MAIN.LOOP.STAGE6
  REM TBD - ELSE, SEND NAK
  PRINT "MISMATCH CRC - ABORTED"
  END

REM STAGE6 - PROCESS RECEIVED BLOCK OF DATA
REM IF FIRST BLOCK, PARSE HEADER (GET FILENAME, LENGTH, ETC.)
REM ELSE, APPEND BLOCK CONTENT TO THE CURRENT FILE
MAIN.LOOP.STAGE6:
  IN.BUFFER.POS = IN.BUFFER.START.ADDRESS : REM RESET WORKING BUFFER POSITION INDEX BACK TO BEGINNING
  IF CURR.BLOCK = 0 THEN GOTO MAIN.LOOP.STAGE6.A : REM TBD - OPEN THE FILE, ESTABLISH HANDLE
  REM ELSE...
  REM TBD -- FILE MANAGEMENT STUFF (APPEND BLOCK TO CURRENT FILE,
  REM UP TO THE NUMBER OF EXPECTED BYTES IN THIS FILE)
  RECEIVED.BYTES = RECEIVED.BYTES+BLOCK.SIZE%
  IF (RECEIVED.BYTES > BLOCK.HEADER.LEN) THEN RECEIVED.BYTES = BLOCK.HEADER.LEN : REM THE REMAINING BYTES ARE IN THIS BLOCK...
  PRINT "5:(APPEND TO [";BLOCK.HEADER.FN$;"], TOTAL ";RECEIVED.BYTES;" BYTES)"
  GOTO MAIN.LOOP.STAGE7

REM WARNING: POST-PARSING BLOCK MAY CAUSE A "STALL" HERE.
REM COULD BE ISSUE WITH SINGLE BYTE UART.  WE HAVE TO PARSE ALL THIS
REM BEFORE THE FIFO BECOMES FULL (THOUGH CTS/RTS FLOW CONTROL SHOULD PROVIDE
REM SOME PROTECTION).
MAIN.LOOP.STAGE6.A:
  REM PARSE HEADER IN THE STORED BLOCK IN RAM
  REM PART 1: <FILENAME> <NUL>
  
  REM NOTE, WE ARE RE-USING SERIAL.IN% EVEN THOUGH THIS IS NOT REALLY A SERIAL INPUT.  IT'S COPY OF RAM BUFFER.
  REM NOTE, THE FOLLOWING ASSUMES THE FILENAME IS AT LEAST 1 CHARACTER.
  SERIAL.IN% = PEEK(IN.BUFFER.POS) : IN.BUFFER.POS = IN.BUFFER.POS+1
  IF SERIAL.IN% = 0 THEN PRINT "6A:HEADER FN [{CYAN}";BLOCK.HEADER.FN$;"{WHITE}] "; : I = 0 : GOTO MAIN.LOOP.STAGE6.B : REM FOUND STRING NUL TERMINATOR
  BLOCK.HEADER.FN$ = BLOCK.HEADER.FN$ + CHR$(SERIAL.IN%)
  GOTO MAIN.LOOP.STAGE6.A : REM GET NEXT FILENAME CHARACTER

MAIN.LOOP.STAGE6.B:
  REM PART 2: <SIZE IN ASCII> <SPACE>
  
  REM NOTE, WE ARE RE-USING SERIAL.IN% EVEN THOUGH THIS IS NOT REALLY A SERIAL INPUT.  IT'S COPY OF RAM BUFFER.
  REM NOTE, FOR 0-BYTE, THE BYTE LENGTH AND OTHER HEADER CONTENT MIGHT BE NUL.  
  SERIAL.IN% = PEEK(IN.BUFFER.POS) : IN.BUFFER.POS = IN.BUFFER.POS+1 : I=I+1
  IF I > 9 THEN BLOCK.HEADER.LEN$ = "0" : SERIAL.IN% = 32 : REM SAFETY PROTECTION
  IF SERIAL.IN% = 32 THEN PRINT "6B:LEN [{RED}";BLOCK.HEADER.LEN$;"{WHITE}]" : GOTO MAIN.LOOP.STAGE6.C
  BLOCK.HEADER.LEN$ = BLOCK.HEADER.LEN$ + CHR$(SERIAL.IN%)
  GOTO MAIN.LOOP.STAGE6.B : REM GET NEXT FILE LENGTH CHARACTER
  
MAIN.LOOP.STAGE6.C:  
  BLOCK.HEADER.LEN = VAL(BLOCK.HEADER.LEN$)
  
  REM TBD - HANDLE THESE LATER...
  REM PART 3: <TIME MODIFIED IN ASCII/OCTAL> <SPACE>
  REM PART 4: <MODE> <SPACE>
  REM PART 5: <SERIAL NUMBER> <SPACE>
  REM PART 6: <OTHER> <SPACE>
  REM PART 7: <OPTIONAL FIELDS>
  REM PART 8: END IS NUMBER OF 128 BYTE BLOCKS
  
  IF LEN(BLOCK.HEADER.FN$) = 0 AND (BLOCK.HEADER.LEN = 0) THEN GOTO PRESUME.FINAL.EOT

REM STAGE7 - SEND ACK AND "C" (CONTINUE) RESPONSE
MAIN.LOOP.STAGE7:
  PRINT "7:ACK  ";
  A$ = CHR$(6) : GOSUB SEND.ONE
  IF CURR.BLOCK = 0 THEN A$ = "C" : GOSUB SEND.ONE : GOTO MAIN.LOOP.STAGE8
  REM IF RECEIVED.BYTES < BLOCK.HEADER.LEN THEN A$ = "C" : GOSUB SEND.ONE
  REM A$ = "C" : GOSUB SEND.ONE

REM STAGE8 - WAIT FOR EOT RESPONSE
REM EOT = ASCII 4 (NOTHING LEFT FOR THIS FILE STREAM)
REM OR - ANOTHER DATA BLOCK (#01 OR #02, +1, -1)
MAIN.LOOP.STAGE8:
  IF CURR.EXPECTED.MARKER1% = $FF THEN CURR.EXPECTED.MARKER1% = $00 : CURR.EXPECTED.MARKER2% = $FF : GOTO SKIP.INCREMENT
  REM ELSE...
  CURR.EXPECTED.MARKER1% = CURR.EXPECTED.MARKER1%+1
  CURR.EXPECTED.MARKER2% = CURR.EXPECTED.MARKER2%-1
SKIP.INCREMENT:  
  CURR.BLOCK = CURR.BLOCK + 1
  CRC.RESULT = 0
  IN.BUFFER.POS = IN.BUFFER.START.ADDRESS  
  GOTO MAIN.LOOP.STAGE1

REM HANDLE "END OF TRANSMISSION" BY ACKNOWLEDGING RECEIPT,
REM AND ISSUE A "CONTINUE" FOR ANY NEXT BATCHED FILE.
HANDLE.EOT:
  PRINT "<RECV EOT>"

  REM RESET BLOCK MANAGEMENT VARIABLES
  CURR.EXPECTED.MARKER1% = $00
  CURR.EXPECTED.MARKER2% = $FF
  CURR.BLOCK = 0
  CRC.RESULT = 0
  IN.BUFFER.POS = IN.BUFFER.START.ADDRESS  
  
  BLOCK.HEADER.FN$ = ""  
  BLOCK.HEADER.LEN$ = ""
  BLOCK.HEADER.LEN = 0
  RECEIVED.BYTES = 0

  A$ = CHR$(6) : GOSUB SEND.ONE : REM "ACK"
  A$ = "C"     : GOSUB SEND.ONE : REM "CONTINUE"
  RETURN
  
PRESUME.FINAL.EOT:
  PRINT "6C:[ BLANK FILE, 0 LEN == END OF BATCH ]"
  REM SEND FINAL ACK
  A$ = CHR$(6) : GOSUB SEND.ONE
  A$ = CHR$(6) : GOSUB SEND.ONE
  END

WAIT.LINE.QUIET:
  REM WAIT FOR HALF A SECOND OF SILENCE ON THE SERIAL RECEIVE LINE
  START.TIME = TI
  PRINT "WAIT RECV DRAIN [";
WAIT.LINE.QUIET.2:
  DELTA.TIME = TI - START.TIME
  SERIAL.IN% = PEEK(REG.READ.RBR)  
  IF (SERIAL.IN% = 0) AND (DELTA.TIME > 300) THEN PRINT "DONE]":RETURN
  IF (SERIAL.IN% <> 0) THEN PRINT HEX$(SERIAL.IN%);" ";:START.TIME = TI
  GOTO WAIT.LINE.QUIET.2  

SEND.ONE:
  A% = PEEK(REG.READ.MSR) 
  IF (A% AND $10) THEN GOTO PROCEED.SEND.ONE : REM STILL CLEAR TO SEND
  PRINT "X";
  GOTO SEND.ONE : REM NOT CLEAR TO SEND, WAIT...
PROCEED.SEND.ONE:  
  A% = ASC(A$)
  POKE REG.WRITE.THR, A%
  PRINT "[";HEX$(A%);"]";
  RETURN

SERIAL.INIT:
  POKE REG.WRITE.LCR,%10000011 : REM SET DLAB==1 (AND SET 8 DATA BIT, 1 STOP BIT)
  REM                %10000010 : REM 7 DATA BIT, 1 STOP BIT
  REM                %10000101 : REM 6 DATA BIT, 2 STOP BIT
  REM                 STOP-^ (0 = 1 STOP BIT, 1 = 2 STOP BIT)
  
  POKE REG.WRITE.DLL,BAUD.DIV.LO(BAUD.CURR) : REM LSB (LO)
  POKE REG.WRITE.DLM,BAUD.DIV.HI(BAUD.CURR) : REM MSB (HI)  

  X = PEEK(REG.READ.LCR) : REM STORE CURRENT BIT MASK
  X = X AND %01111111    : REM SET LCR.DLAB BIT == 0
  POKE REG.WRITE.LCR, X

  POKE REG.WRITE.FCR,%10000111 : REM TRIGGER AT 8 BYTE FIFO, CLEAR TX/RX FIFO BUFFERS, ENABLE FIFO

  POKE REG.WRITE.MCR, %00100001 : REM SET DTR

  REM BAUD RATE == BAUD.RATE$(BAUD.CURR)

  RETURN

INIT.SERIAL.REGS:
  REM WHEN DLAB==0 (BIT 7 OF REG.BASE+3)
  REM --------------------
  REG.READ.RBR = SERIAL.REG.BASE+0 : REM RECEIVER BUFFER REGISTER (READ-ONLY)
  REG.READ.IER = SERIAL.REG.BASE+1 : REM INTERRUPT ENABLE REGISTER
  REG.READ.IIR = SERIAL.REG.BASE+2 : REM INTERRUPT IDENT REGISTER (READ-ONLY)
  REG.READ.LCR = SERIAL.REG.BASE+3 : REM LINE CONTROL REGISTER
  REG.READ.MCR = SERIAL.REG.BASE+4 : REM MODEM CONTROL REGISTER
  REG.READ.LSR = SERIAL.REG.BASE+5 : REM LINE STATUS REGISTER
  REG.READ.MSR = SERIAL.REG.BASE+6 : REM MODEM STATUS REGISTER
  REG.READ.SCR = SERIAL.REG.BASE+7 : REM SCRATCH REGISTER  

  REG.WRITE.THR = SERIAL.REG.BASE+0 : REM TRANSMIT HOLDING REGISTER (WRITE-ONLY)
  REG.WRITE.FCR = SERIAL.REG.BASE+2 : REM FIFO CONTROL REGISTER (WRITE-ONLY)
  REG.WRITE.LCR = SERIAL.REG.BASE+3 : REM LINE CONTROL REGISTER
  REG.WRITE.MCR = SERIAL.REG.BASE+4 : REM MODEM CONTROL REGISTER
  REG.WRITE.NA1 = SERIAL.REG.BASE+5 : REM LINE STATUS REGISTER
  REG.WRITE.NA2 = SERIAL.REG.BASE+6 : REM MODEM STATUS REGISTER
  REG.WRITE.SCR = SERIAL.REG.BASE+7 : REM SCRATCH REGISTER

  REM WHEN DLAB==1 (BIT 7 OF SERIAL.REG.BASE+3 LCR, DIVISOR LATCH ACCESS BIT)
  REM --------------------
  REG.READ.DLL = SERIAL.REG.BASE+0 : REM DIVISOR LATCH LSB
  REG.READ.DLM = SERIAL.REG.BASE+1 : REM DIVISOR LATCH MSB
  REM  IIR,LCR,MCR SAME
  REM  LSR,MSR SAME

  REG.WRITE.DLL = SERIAL.REG.BASE+0 : REM DIVISOR LATCH LSB
  REG.WRITE.DLM = SERIAL.REG.BASE+1 : REM DIVISOR LATCH MSB
  REM  FCR,LCR,MCR SAME
  RETURN

INIT.SERIAL.ADDR:
  REM ASSUME DEVICE.MODE == 1 (SERIAL)
  REM LOW ADDRESSES
  SERIAL.ADDR(3) = $9F68
  SERIAL.ADDR(4) = $9F88
  SERIAL.ADDR(5) = $9FA8
  SERIAL.ADDR(6) = $9FC8
  SERIAL.ADDR(7) = $9FE8
  REM HIGH ADDRESSES
  SERIAL.ADDR(10) = $9F78
  SERIAL.ADDR(11) = $9F98
  SERIAL.ADDR(12) = $9FB8
  SERIAL.ADDR(13) = $9FD8
  SERIAL.ADDR(14) = $9FF8
  IF DEVICE.MODE = 1 THEN RETURN : REM STICK WITH THE SERIAL SIDE DEFAULTS

  REM ELSE, OVERRIDE SERIAL.ADDR WITH THE "NETWORK" SIDE PORTS
  SERIAL.ADDR(3) = $9F60
  SERIAL.ADDR(4) = $9F80
  SERIAL.ADDR(5) = $9FA0
  SERIAL.ADDR(6) = $9FC0
  SERIAL.ADDR(7) = $9FE0
  REM HIGH ADDRESSES
  SERIAL.ADDR(10) = $9F70
  SERIAL.ADDR(11) = $9F90
  SERIAL.ADDR(12) = $9FB0
  SERIAL.ADDR(13) = $9FD0
  SERIAL.ADDR(14) = $9FF0
  RETURN
